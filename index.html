// Function to fetch and display items function loadItems() { const itemsRef = database.ref('products/items'); itemsRef.on('value', (snapshot) => { // Check if snapshot has data before clearing and appending if (snapshot.exists()) { itemsListDiv.innerHTML = ''; // Clear loading message ONLY if data exists snapshot.forEach((childSnapshot) => { const item = childSnapshot.val(); const itemId = childSnapshot.key; const itemElement = document.createElement('div'); itemElement.className = 'product-card'; itemElement.innerHTML = `

${item.name}

${item.price.toLocaleString()} تومان

`; itemsListDiv.appendChild(itemElement); }); } else { itemsListDiv.innerHTML = '

آیتمی برای نمایش وجود ندارد.

'; // Message if no items found } }, (error) => { console.error("Error loading items:", error); itemsListDiv.innerHTML = '

خطا در بارگذاری آیتم‌ها.

'; }); }
