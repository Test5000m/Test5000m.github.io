// --- تنظیمات اولیه --- const itemsListDiv = document.getElementById('items-list'); const ranksListDiv = document.getElementById('ranks-list'); // --- تابع برای نمایش قیمت با فرمت تومان --- function formatPrice(price) { return price.toLocaleString('fa-IR') + ' تومان'; // استفاده از 'fa-IR' برای فرمت فارسی } // --- تابع برای نمایش آیتم‌ها (شامل رنک‌ها و آیتم‌های دیگر) --- function loadProductSection(refPath, targetDiv, productType) { const productRef = database.ref(refPath); const loadingDiv = targetDiv.querySelector('.loading-message'); if (loadingDiv) loadingDiv.remove(); // حذف پیام بارگذاری اولیه console.log(`Fetching data from: ${refPath}`); productRef.on('value', (snapshot) => { console.log(`Data received for ${refPath}. Exists: ${snapshot.exists()}`); if (snapshot.exists()) { targetDiv.innerHTML = ''; // پاک کردن محتوای قبلی snapshot.forEach((childSnapshot) => { const product = childSnapshot.val(); const productId = childSnapshot.key; // اطمینان از وجود نام و قیمت قبل از ایجاد عنصر if (product && typeof product.name !== 'undefined' && typeof product.price !== 'undefined') { const productElement = document.createElement('div'); productElement.className = 'product-card'; // اضافه کردن جزئیات بیشتر بر اساس نوع محصول let detailsHtml = ''; if (product.description) { detailsHtml += `

توضیحات: ${product.description}

`; } if (product.type) { detailsHtml += `

نوع: ${product.type}

`; } if (product.effect) { detailsHtml += `

تأثیر: ${product.effect}

`; } // دکمه مربوط به هر محصول (مثلا اضافه کردن به سبد) const buttonAction = productType === 'rank' ? 'selectRank' : 'addItemToCart'; const buttonText = productType === 'rank' ? 'انتخاب رنک' : 'اضافه کردن'; productElement.innerHTML = `

${product.name}

${formatPrice(product.price)}

${detailsHtml} `; targetDiv.appendChild(productElement); console.log(`Rendered ${productType}: ${product.name} (ID: ${productId})`); } else { console.warn(`Skipping invalid product data. Path: ${refPath}, Key: ${productId}`, product); } }); if (targetDiv.children.length === 0) { targetDiv.innerHTML = '
موردی برای نمایش وجود ندارد.
'; } } else { targetDiv.innerHTML = '
هیچ موردی در این دسته یافت نشد.
'; console.log(`No items found at ${refPath}`); } }, (error) => { console.error(`Error loading ${refPath}:`, error); targetDiv.innerHTML = '
خطا در بارگذاری اطلاعات. لطفاً بعداً تلاش کنید.
'; }); } // --- توابع مربوط به دکمه‌ها (می‌توانید این‌ها را گسترش دهید) --- function selectRank(rankId, rankName, rankPrice) { console.log(`Rank selected: ${rankName} (ID: ${rankId}, Price: ${rankPrice})`); alert(`شما رنک ${rankName} را انتخاب کردید! (قیمت: ${formatPrice(rankPrice)})`); // اینجا می‌توانید منطق اضافه کردن رنک به سبد خرید یا اعمال آن را اضافه کنید } function addItemToCart(itemId, itemName, itemPrice) { console.log(`Item added to cart: ${itemName} (ID: ${itemId}, Price: ${itemPrice})`); alert(`آیتم ${itemName} به سبد خرید شما اضافه شد!`); // اینجا می‌توانید منطق اضافه کردن آیتم به سبد خرید واقعی را پیاده‌سازی کنید // مثلاً: updateCart(itemId, itemName, itemPrice); } // --- بارگذاری اولیه بخش‌ها --- document.addEventListener('DOMContentLoaded', () => { // بارگذاری رنک‌ها loadProductSection('ranks', ranksListDiv, 'rank'); // بارگذاری آیتم‌های کلیدی loadProductSection('items', itemsListDiv, 'item'); });
